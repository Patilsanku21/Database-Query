create procedure sp_studdents_information_by_S_id
@S_id int = 1
AS
BEGIN
select S_id,name,Age, salary from students where S_id = @S_id;
END;

create procedure sp_students_information_salary_Range
@start money,
@END money
AS
BEGIN
select * from students where salary between @start and @END;

END;
